<?php

namespace App\Console\Commands;

use App\Models\Product;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\File;
use Illuminate\Support\Str;

class CheckProductPhotos extends Command
{
    protected $signature = 'catalogue:check {--path= : Folder that holds the photos}'; // php artisan catalogue:check
    protected $description = 'Compare product photos on disk with the products and colours in the database';

    private const EXTENSIONS = ['png', 'jpg', 'jpeg', 'webp'];   // allowed, lowercase only

    public function handle(): int
    {
        $dir = rtrim($this->option('path') ?: base_path('../frontend/public/products'), '/\\'); // where to look

        if (! is_dir($dir)) {
            $this->components->error("Folder not found: {$dir}");
            return self::FAILURE;
        }

        $have = collect(File::files($dir))->map(fn ($f) => $f->getFilename())->all(); // names on disk

        // base name (no extension) => full file name, only for allowed lowercase extensions
        $byBase = [];
        foreach ($have as $file) {
            if (in_array(pathinfo($file, PATHINFO_EXTENSION), self::EXTENSIONS, true)) {
                $byBase[pathinfo($file, PATHINFO_FILENAME)] = $file;
            }
        }

        $expected = [];      // every base name the catalogue can use
        $missingColor = [];  // colours with no photo
        $noPhoto = [];       // products that will show the placeholder
        $colorTotal = 0;

        foreach (Product::orderBy('id')->get() as $product) {
            $expected[] = $product->slug;                          // optional main photo
            $anyPhoto = isset($byBase[$product->slug]);            // main photo present?

            foreach ($product->colors as $color) {
                $base = $product->slug . '-' . Str::slug($color['name']);
                $expected[] = $base;
                $colorTotal++;
                if (isset($byBase[$base])) {
                    $anyPhoto = true;
                } else {
                    $missingColor[] = $base . '.png';
                }
            }

            if (! $anyPhoto) {
                $noPhoto[] = $product->slug;
            }
        }

        // Files that match nothing (gallery extras like name-2.png are allowed)
        $unknown = collect($have)->reject(function ($f) use ($expected, $byBase) {
            $base = pathinfo($f, PATHINFO_FILENAME);
            return $f === 'placeholder.svg'
                || (isset($byBase[$base]) && (in_array($base, $expected, true) || preg_match('/^[a-z0-9-]+-[2-6]$/', $base)));
        })->values();

        $found = $colorTotal - count($missingColor);
        $this->components->info(count($have) . " files found. Colour photos: {$found} of {$colorTotal}. The black photo is used as the main one.");

        if ($noPhoto) {
            $this->components->warn('Products with NO photo at all (will show the placeholder) (' . count($noPhoto) . '):');
            $this->components->bulletList($noPhoto);
        }
        if ($missingColor) {
            $this->components->warn('Missing COLOUR photos (' . count($missingColor) . '):');
            $this->components->bulletList($missingColor);
        }
        if ($unknown->isNotEmpty()) {
            $this->components->error('Files with a wrong name (' . $unknown->count() . '):');
            $this->components->bulletList($unknown->map(fn ($f) => $this->describe($f, $expected))->all());
        }

        if (! $missingColor && ! $noPhoto && $unknown->isEmpty()) {
            $this->components->info('Perfect: every photo is named correctly.');
            return self::SUCCESS;
        }

        return self::FAILURE;
    }

    // Explains what is wrong with a file name and suggests the closest right name
    private function describe(string $file, array $expected): string
    {
        $lower = strtolower($file);
        $lowerBase = strtolower(pathinfo($file, PATHINFO_FILENAME));

        foreach ($expected as $base) {
            if ($lowerBase === $base) {                      // right name, wrong capitals or extension
                return "{$file}  ->  use lowercase and an allowed extension: {$base}.png";
            }
        }

        $best = null;
        $bestScore = 99;
        foreach ($expected as $base) {
            $score = levenshtein($lowerBase, $base);         // how many characters differ
            if ($score < $bestScore) {
                $best = $base;
                $bestScore = $score;
            }
        }

        return $bestScore <= 4 ? "{$file}  ->  did you mean {$best}.png?" : "{$file}  ->  no matching product";
    }
}


===== .\app\Console\Commands\CreateStaffUser.php =====
<?php

namespace App\Console\Commands;

use App\Enums\UserRole;
use App\Models\User;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Validator;

class CreateStaffUser extends Command
{
    protected $signature = 'staff:create {--role=super_admin : super_admin or admin}'; // php artisan staff:create

    protected $description = 'Create an admin or super admin account from the command line';

    public function handle(): int
    {
        $role = UserRole::tryFrom((string) $this->option('role'));

        if (! $role || ! $role->isStaff()) {
            $this->error('Role must be super_admin or admin.');
            return self::FAILURE;
        }

        $name     = $this->ask('Full name');
        $email    = $this->ask('Email');
        $password = $this->secret('Password (min 12 characters, hidden as you type)');

        $validator = Validator::make(compact('name', 'email', 'password'), [
            'name'     => ['required', 'string', 'max:100'],
            'email'    => ['required', 'email', 'unique:users,email'], // Must not exist already
            'password' => ['required', 'string', 'min:12'],
        ]);

        if ($validator->fails()) {
            foreach ($validator->errors()->all() as $message) {
                $this->error($message);
            }
            return self::FAILURE;
        }

        $user = User::create(compact('name', 'email', 'password')); // Password is hashed by the model cast

        // role is not mass-assignable on purpose, so we set it here with forceFill
        $user->forceFill(['role' => $role, 'is_active' => true])->save();

        $this->info("{$role->label()} created: {$email}");

        return self::SUCCESS;
    }
}

<?php

namespace App\Console\Commands;

use App\Models\Product;
use App\Models\ProductStock;
use App\Services\CatalogueCache;
use Illuminate\Console\Command;

class InitStock extends Command
{
    protected $signature = 'stock:init {--quantity=20 : Starting quantity for each option} {--only= : One product slug}';

    protected $description = 'Create stock rows (default 20 each) for products that have none. Existing rows are not touched.';

    public function handle(CatalogueCache $cache): int
    {
        $quantity = max(0, (int) $this->option('quantity'));
        $created = 0;

        $products = Product::query()->when($this->option('only'), fn ($q, $slug) => $q->where('slug', $slug))->get();

        foreach ($products as $product) {
            foreach ($product->colors ?? [] as $color) {
                foreach ($product->sizes ?? [] as $size) {
                    $row = ProductStock::firstOrCreate(                       // Only creates what is missing
                        ['product_id' => $product->id, 'color' => $color['name'], 'size' => $size],
                        ['quantity' => $quantity]
                    );
                    $created += $row->wasRecentlyCreated ? 1 : 0;
                }
            }
        }

        $cache->flush(); // Public pages pick up the new availability

        $this->info("Created {$created} stock rows with {$quantity} each. Replace these placeholders with real counts in the admin.");

        return self::SUCCESS;
    }
}

<?php

namespace App\Console\Commands;

use App\Models\Product;
use App\Services\StockService;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class StockRaceDemo extends Command
{
    protected $signature = 'stock:race {product : Product id} {color} {size} {--hold=0 : Seconds to keep the lock after taking 1}';

    protected $description = 'DEMO: try to buy 1 unit and hold the lock, to watch a second process wait';

    public function handle(StockService $stock): int
    {
        $product = Product::findOrFail($this->argument('product'));

        try {
            DB::transaction(function () use ($stock, $product) {
                $this->line('Trying to take 1 unit...');
                $stock->reserve([[
                    'product_id' => $product->id, 'name' => $product->name,
                    'color' => $this->argument('color'), 'size' => $this->argument('size'), 'quantity' => 1,
                ]]);
                $this->info('GOT IT. Holding the lock for ' . $this->option('hold') . 's...');
                sleep((int) $this->option('hold'));  // Keep the transaction open so the other window has to wait
            });
            $this->info('COMMITTED: bought 1.');
        } catch (ValidationException $e) {
            $this->error('REFUSED: ' . collect($e->errors())->flatten()->first());
        }

        return self::SUCCESS;
    }
}


===== .\app\Console\Commands\UploadProductImages.php =====
<?php

namespace App\Console\Commands;

use App\Models\Product;
use App\Models\ProductPhoto;
use App\Services\Contracts\ImageUploaderInterface;
use Illuminate\Console\Command;
use Illuminate\Support\Str;
use Throwable;

class UploadProductImages extends Command
{
    protected $signature = 'images:upload {--only= : One product slug} {--force : Re-upload photos already uploaded}';

    protected $description = 'Upload product photos from frontend/public/products to Cloudinary';

    public function handle(ImageUploaderInterface $uploader): int
    {
        $dir = base_path('../frontend/public/products');            // Where the photos live
        $files = [];                                                // base name => full path
        foreach (glob($dir . '/*.{png,jpg,jpeg,webp}', GLOB_BRACE) as $path) {
            $files[pathinfo($path, PATHINFO_FILENAME)] = $path;     // "core-leggings-black" => path
        }

        $products = Product::with('photos')                         // Load existing photo rows once
            ->when($this->option('only'), fn ($q, $slug) => $q->where('slug', $slug))
            ->get();

        $uploaded = $skipped = $failed = 0;                         // Counters for the summary

        foreach ($products as $product) {
            $done = $product->photos                                // Keys of photos already in the DB
                ->map(fn ($p) => ($p->color ?? '') . '|' . $p->position)->all();

            $wanted = [];                                           // [file base, color slug|null, position]
            foreach ($product->colors ?? [] as $color) {
                $colorSlug = Str::slug($color['name']);             // "Dark Blue" => "dark-blue"
                $wanted[] = [$product->slug . '-' . $colorSlug, $colorSlug, 1];
            }
            $wanted[] = [$product->slug, null, 1];                  // Optional main photo
            foreach (range(2, 6) as $n) {
                $wanted[] = [$product->slug . '-' . $n, null, $n];  // Optional gallery photos
            }

            foreach ($wanted as [$base, $color, $position]) {
                if (! isset($files[$base])) {
                    continue;                                       // No such photo, nothing to upload
                }
                if (in_array(($color ?? '') . '|' . $position, $done, true) && ! $this->option('force')) {
                    $skipped++;                                     // Already uploaded earlier
                    continue;
                }

                try {
                    $result = $uploader->upload($files[$base], $base); // Send to Cloudinary
                    ProductPhoto::updateOrCreate(
                        ['product_id' => $product->id, 'color' => $color, 'position' => $position],
                        ['public_id' => $result['public_id'], 'url' => $result['url']]
                    );
                    $uploaded++;
                    $this->line("uploaded  {$base}");
                } catch (Throwable $e) {
                    $failed++;                                      // Keep going, report at the end
                    $this->error("FAILED    {$base}: {$e->getMessage()}");
                }
            }
        }

        $this->info("Done. Uploaded {$uploaded}, skipped {$skipped}, failed {$failed}.");

        return $failed ? self::FAILURE : self::SUCCESS;
    }
}

===== .\app\Enums\OrderStatus.php =====
<?php

namespace App\Enums;

enum OrderStatus: string
{
    case Pending    = 'pending';    // Placed, waiting for payment
    case Paid       = 'paid';       // Payment confirmed
    case Processing = 'processing'; // Being packed
    case Shipped    = 'shipped';    // On its way
    case Delivered  = 'delivered';  // Arrived
    case Cancelled  = 'cancelled';  // Stopped before delivery

    // Friendly text for the website
    public function label(): string
    {
        return match ($this) {
            self::Pending    => 'Awaiting payment',
            self::Paid       => 'Paid',
            self::Processing => 'Processing',
            self::Shipped    => 'Shipped',
            self::Delivered  => 'Delivered',
            self::Cancelled  => 'Cancelled',
        };
    }

        // Where an order may go next (an admin cannot set "paid": only a confirmed payment does)
    public function allowedNext(): array
    {
        return match ($this) {
            self::Pending    => [self::Cancelled],
            self::Paid       => [self::Processing, self::Cancelled],
            self::Processing => [self::Shipped, self::Cancelled],
            self::Shipped    => [self::Delivered],
            self::Delivered, self::Cancelled => [], // Final states
        };
    }

    public function canMoveTo(self $next): bool
    {
        return in_array($next, $this->allowedNext(), true);
    }
}

<?php

namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Requests\LoginRequest;
use App\Http\Requests\RegisterRequest;
use App\Http\Resources\UserResource;
use App\Services\AuthService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Response;

class AuthController extends Controller
{
    public function __construct(private AuthService $auth) {}

    // POST /api/v1/register
    public function register(RegisterRequest $request): JsonResponse
    {
        $result = $this->auth->register($request->toData());

        return response()->json([
            'user'  => new UserResource($result['user']),
            'token' => $result['token'],
        ], 201); // 201 = created
    }

    // POST /api/v1/login
    public function login(LoginRequest $request): JsonResponse
    {
        $result = $this->auth->login($request->toData());

        return response()->json([
            'user'  => new UserResource($result['user']),
            'token' => $result['token'],
        ]);
    }

    // GET /api/v1/me (needs a token)
    public function me(Request $request): JsonResponse
    {
        return response()->json(['user' => new UserResource($request->user())]);
    }

    // POST /api/v1/logout (needs a token)
    public function logout(Request $request): Response
    {
        $this->auth->logout($request->user());

        return response()->noContent(); // 204 = done, nothing to return
    }
}

<?php

namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Requests\CheckoutRequest;
use App\Http\Requests\OrderAccessRequest;
use App\Http\Resources\OrderResource;
use App\Services\OrderService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\AnonymousResourceCollection;

class OrderController extends Controller
{
    public function __construct(private OrderService $orders) {}

    // POST /api/v1/orders (place an order: guest or signed in)
    public function store(CheckoutRequest $request): JsonResponse
    {
        $order = $this->orders->place($request->user('sanctum'), $request->toData(), $request->idempotencyKey()); // null user = guest

        // 201 = newly created, 200 = a repeat of an earlier attempt (same order returned)
        return (new OrderResource($order))->response()->setStatusCode($order->wasRecentlyCreated ? 201 : 200);
    }

    // POST /api/v1/orders/{reference}/lookup (guest proves ownership with the order's email)
    public function lookup(OrderAccessRequest $request, string $reference): OrderResource
    {
        return new OrderResource($this->orders->findFor($request->user('sanctum'), $reference, $request->email()));
    }

    // GET /api/v1/orders (my orders)
    public function index(Request $request): AnonymousResourceCollection
    {
        return OrderResource::collection($this->orders->listFor($request->user()));
    }

    // GET /api/v1/orders/{reference} (one of my orders)
    public function show(Request $request, string $reference): OrderResource
    {
        return new OrderResource($this->orders->findFor($request->user(), $reference));
    }
}

<?php

namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Requests\OrderAccessRequest;
use App\Http\Resources\OrderResource;
use App\Services\PaymentService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class PaymentController extends Controller
{
    public function __construct(private PaymentService $payments) {}

    // POST /api/v1/orders/{reference}/pay (get the Paystack page URL)
    public function pay(OrderAccessRequest $request, string $reference): JsonResponse
    {
        return response()->json([
            'authorizationUrl' => $this->payments->start($request->user('sanctum'), $reference, $request->email()),
        ]);
    }

    // POST /api/v1/orders/{reference}/confirm (customer is back, check the payment)
    public function confirm(OrderAccessRequest $request, string $reference): OrderResource
    {
        return new OrderResource($this->payments->confirm($request->user('sanctum'), $reference, $request->email()));
    }

    public function webhook(Request $request): JsonResponse
    {
        $this->payments->handleWebhook($request->getContent(), $request->header('x-paystack-signature')); // Raw body is needed for the signature

        return response()->json(['ok' => true]);
    }
}

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureStaff
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = $request->user();

        if (! $user || ! $user->isStaff() || ! $user->is_active) {
            abort(403, 'Staff access only.'); // 403 = logged in, but not allowed
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Requests;

use App\DTOs\CartLineData;
use App\DTOs\CheckoutData;
use Illuminate\Foundation\Http\FormRequest;

class CheckoutRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true; // Open to guests and signed-in customers
    }

    public function rules(): array
    {
        return [
            'email'              => ['required', 'email:rfc', 'max:255'], // Receipt + payment email
            'recipient_name'     => ['required', 'string', 'max:100'],
            'phone'              => ['required', 'string', 'max:20'],
            'address'            => ['required', 'string', 'max:255'],
            'city'               => ['required', 'string', 'max:100'],
            'state'              => ['required', 'string', 'max:100'],
            'items'              => ['required', 'array', 'min:1', 'max:30'],
            'items.*.product_id' => ['required', 'integer', 'exists:products,id'],
            'items.*.color'      => ['required', 'string', 'max:50'],
            'items.*.size'       => ['required', 'string', 'max:50'],
            'items.*.quantity'   => ['required', 'integer', 'min:1', 'max:' . config('shop.max_qty')],
            'idempotency_key'    => ['nullable', 'string', 'regex:/^[A-Za-z0-9_-]{8,64}$/'], // From the header
            // No price field on purpose: the server decides prices
        ];
    }

    // Turn the validated input into typed objects
    public function toData(): CheckoutData
    {
        $v = $this->validated();

        return new CheckoutData(
            recipientName: trim($v['recipient_name']),
            email: mb_strtolower(trim($v['email'])), // Lowercase so lookups match
            phone: trim($v['phone']),
            address: trim($v['address']),
            city: trim($v['city']),
            state: trim($v['state']),
            lines: array_map(
                fn (array $i) => new CartLineData((int) $i['product_id'], $i['color'], $i['size'], (int) $i['quantity']),
                $v['items']
            ),
        );
    }

    // Copy the header into the input, and force the account email for signed-in customers
    protected function prepareForValidation(): void
    {
        $this->merge([
            'idempotency_key' => $this->header('Idempotency-Key'),
            ...($this->user('sanctum') ? ['email' => $this->user('sanctum')->email] : []),
        ]);
    }

    // The validated key, or null if the client didn't send one
    public function idempotencyKey(): ?string
    {
        return $this->validated('idempotency_key');
    }
}

===== .\app\Http\Resources\ProductResource.php =====
<?php

namespace App\Http\Resources;

use App\Models\ProductStock;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;
use Illuminate\Support\Str;

class ProductResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        $local = $this->images ?? [];                                   // Old local photo paths (fallback)
        $photos = $this->relationLoaded('photos') ? $this->photos : collect(); // Cloudinary rows
        $byColor = $photos->whereNotNull('color')->pluck('url', 'color');       // "black" => url
        $gallery = $photos->whereNull('color')->sortBy('position')->pluck('url')->values()->all();

        $colors = collect($this->colors ?? [])->map(fn ($c) => [
            ...$c,                                                      // name, hex, old image
            'image' => $byColor[Str::slug($c['name'])] ?? ($c['image'] ?? null), // Cloudinary first
        ])->all();

        $main = $gallery[0] ?? $byColor['black'] ?? $byColor->first()   // Main, else black, else any colour
            ?? ($local[0] ?? '/products/placeholder.svg');

        [$soldOutByColor, $soldOutSizes, $allSoldOut] = $this->availability();

        return [
            'id'             => $this->id,
            'slug'           => $this->slug,
            'name'           => $this->name,
            'category'       => $this->whenLoaded('category', fn () => $this->category->name),
            'price'          => $this->price,
            'image'          => $main,
            'gallery'        => array_slice($gallery, 1),               // Photos after the main one
            'badge'          => $this->badge?->value,                   // enum -> plain text, or null
            'description'    => $this->description,
            'features'       => $this->features ?? [],
            'fabric'         => $this->fabric,
            'care'           => $this->care,
            'sizeLabel'      => $this->size_label?->value,              // enum -> plain text
            'sizes'          => $this->sizes,
            'soldOut'        => $soldOutSizes,                          // Sizes unavailable in EVERY colour
            'soldOutByColor' => (object) $soldOutByColor,               // {"Black": ["XL"], ...} (counts are never exposed)
            'colors'         => $colors,
            'isSoldOut'      => $this->is_sold_out || $allSoldOut,      // Manual switch, or nothing left at all
            'isFeatured'     => $this->is_featured,
        ];
    }

    // Which options can't be bought right now: [by colour, sold-out sizes, everything sold out?]
    private function availability(): array
    {
        $manual = $this->sold_out_sizes ?? [];                          // Sizes an admin switched off by hand

        if (! $this->relationLoaded('stocks')) {
            return [[], $manual, false];                                // Stock not loaded: fall back to the manual flags
        }

        $qty = $this->stocks->mapWithKeys(                              // key => quantity
            fn ($s) => [ProductStock::key($s->product_id, $s->color, $s->size) => $s->quantity]
        );

        $colors = $this->colors ?? [];
        $sizes  = $this->sizes ?? [];
        $byColor = [];
        $outCount = 0;

        foreach ($colors as $c) {
            foreach ($sizes as $size) {
                $out = ($qty[ProductStock::key($this->id, $c['name'], $size)] ?? 0) <= 0 || in_array($size, $manual, true);

                if ($out) {
                    $byColor[$c['name']][] = $size;
                    $outCount++;
                }
            }
        }

        $soldOutSizes = array_values(array_filter($sizes, function ($size) use ($byColor, $colors) {
            return $colors !== [] && count(array_filter($byColor, fn ($list) => in_array($size, $list, true))) === count($colors);
        }));

        return [$byColor, $soldOutSizes, $colors !== [] && $outCount === count($colors) * count($sizes)];
    }
}

<?php

namespace App\Repositories\Eloquent;

use App\Enums\OrderStatus;
use App\Repositories\Contracts\AnalyticsRepositoryInterface;
use Carbon\CarbonInterface;
use Illuminate\Database\Query\Builder;
use Illuminate\Support\Facades\DB;

class AnalyticsRepository implements AnalyticsRepositoryInterface
{
    // Orders that count as real sales
    private function paidOrders(CarbonInterface $from, CarbonInterface $to): Builder
    {
        $statuses = array_map(fn (OrderStatus $s) => $s->value, [
            OrderStatus::Paid, OrderStatus::Processing, OrderStatus::Shipped, OrderStatus::Delivered,
        ]);

        return DB::table('orders')
            ->whereIn('status', $statuses)
            ->where('created_at', '>=', $from)
            ->where('created_at', '<', $to);
    }

    public function totals(CarbonInterface $from, CarbonInterface $to): array
    {
        $row = $this->paidOrders($from, $to)
            ->selectRaw('COALESCE(SUM(total), 0) as revenue, COUNT(*) as orders')
            ->first();

        return ['revenue' => (int) $row->revenue, 'orders' => (int) $row->orders];
    }

    public function daily(CarbonInterface $from, CarbonInterface $to, int $offsetMinutes): array
    {
        return $this->paidOrders($from, $to)
            ->selectRaw("DATE(DATE_ADD(created_at, INTERVAL {$offsetMinutes} MINUTE)) as day, COALESCE(SUM(total), 0) as revenue, COUNT(*) as orders") // Shift to Lagos time, then take the date
            ->groupBy('day')
            ->get()->keyBy('day')->all();
    }

    public function statusCounts(): array
    {
        return DB::table('orders')
            ->selectRaw('status, COUNT(*) as total')
            ->groupBy('status')
            ->pluck('total', 'status')->all();
    }

    public function topProducts(CarbonInterface $from, CarbonInterface $to, int $limit): array
    {
        $statuses = array_map(fn (OrderStatus $s) => $s->value, [
            OrderStatus::Paid, OrderStatus::Processing, OrderStatus::Shipped, OrderStatus::Delivered,
        ]);

        return DB::table('order_items')
            ->join('orders', 'orders.id', '=', 'order_items.order_id')
            ->whereIn('orders.status', $statuses)
            ->where('orders.created_at', '>=', $from)
            ->where('orders.created_at', '<', $to)
            ->selectRaw('order_items.product_id, MAX(order_items.name) as name, SUM(order_items.quantity) as units, SUM(order_items.quantity * order_items.unit_price) as revenue')
            ->groupBy('order_items.product_id')
            ->orderByDesc('units')->orderByDesc('revenue')
            ->limit($limit)
            ->get()->all();
    }
}

<?php

namespace App\Repositories\Eloquent;

use App\DTOs\OrderFilterData;
use App\Enums\OrderStatus;
use App\Models\Order;
use App\Models\OrderStatusHistory;
use App\Models\User;
use App\Repositories\Contracts\OrderRepositoryInterface;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class OrderRepository implements OrderRepositoryInterface
{
    public function create(?User $user, array $attributes, array $items): Order
    {
        return DB::transaction(function () use ($user, $attributes, $items) { // If anything fails, nothing is saved
            $order = Order::create([
                ...$attributes,
                'user_id'   => $user?->id,         // null = guest
                'reference' => $this->newReference(),
            ]);

            $order->items()->createMany($items); // One row per cart line
            $order->statusHistories()->create([
                'user_id' => $user?->id, 'from_status' => null, 'to_status' => $order->status,
                'note' => $user ? 'Order placed' : 'Order placed (guest checkout)',
            ]); // First entry in the timeline

            return $order->load('items');
        });
    }

    public function forUser(User $user): Collection
    {
        return Order::with('items')->where('user_id', $user->id)->latest()->get();
    }

    public function findForUser(User $user, string $reference): ?Order
    {
        return Order::with('items')
            ->where('user_id', $user->id)   // Never show someone else's order
            ->where('reference', $reference)
            ->first();
    }

    public function findForCustomer(?User $user, ?string $email, string $reference): ?Order
    {
        if ($user) {
            return $this->findForUser($user, $reference);               // Account owner
        }

        if (! $email) {
            return null;                                                // A guest must give the email
        }

        return Order::with('items')
            ->whereNull('user_id')                                      // Guest orders only
            ->where('reference', $reference)
            ->where('email', $email)                                    // Wrong email = same as "not found"
            ->first();
    }

    // A short readable code like BT-7K3M9QXA, checked for uniqueness
    private function newReference(): string
    {
        do {
            $reference = 'BT-' . strtoupper(Str::random(8));
        } while (Order::where('reference', $reference)->exists());

        return $reference;
    }

    public function findByReference(string $reference): ?Order
    {
        return Order::with('items')->where('reference', $reference)->first();
    }

    public function attachPaymentReference(Order $order, string $paymentReference): void
    {
        $order->update(['payment_reference' => $paymentReference]);
    }

    public function markPaid(Order $order, string $paymentReference): bool
    {
        return DB::transaction(function () use ($order, $paymentReference) { // Status + history saved together
            $changed = Order::where('id', $order->id)
                ->where('status', OrderStatus::Pending->value)   // Only if still pending, so a double call does nothing
                ->update([
                    'status'            => OrderStatus::Paid->value,
                    'payment_reference' => $paymentReference,
                    'paid_at'           => now(),
                ]) === 1;

            if ($changed) {
                OrderStatusHistory::create([
                    'order_id' => $order->id, 'user_id' => null, // null = done by the system
                    'from_status' => OrderStatus::Pending, 'to_status' => OrderStatus::Paid,
                    'note' => 'Payment confirmed by Paystack',
                ]);
            }

            return $changed;
        });
    }

    public function findByIdempotencyKey(?User $user, string $email, string $key): ?Order
    {
        $query = Order::with('items')->where('idempotency_key', $key);

        return ($user
            ? $query->where('user_id', $user->id)                       // Keys are per customer
            : $query->whereNull('user_id')->where('email', $email)      // Guests: per email
        )->first();
    }

    public function countPendingForEmail(string $email): int
    {
        return Order::where('email', $email)->where('status', OrderStatus::Pending)->count();
    }

    public function paginateForAdmin(OrderFilterData $filters): LengthAwarePaginator
    {
        return Order::with(['items', 'user'])
            ->when($filters->status, fn ($q, $status) => $q->where('status', $status->value)) // By stage
            ->when($filters->search, function ($q, $term) {
                $q->where(function ($w) use ($term) {                                          // Grouped, so the ORs stay together
                    $w->where('reference', 'like', "%{$term}%")
                      ->orWhere('recipient_name', 'like', "%{$term}%")
                      ->orWhere('phone', 'like', "%{$term}%")
                      ->orWhere('email', 'like', "%{$term}%");                                 // Works for guests too
                });
            })
            ->when($filters->from, fn ($q, $date) => $q->whereDate('created_at', '>=', $date)) // From date
            ->when($filters->to, fn ($q, $date) => $q->whereDate('created_at', '<=', $date))   // To date
            ->latest()
            ->paginate($filters->perPage);
    }

    public function findForAdmin(string $reference): ?Order
    {
        return Order::with(['items', 'user', 'statusHistories.user'])->where('reference', $reference)->first();
    }

    public function lockByReference(string $reference): ?Order
    {
        return Order::where('reference', $reference)->lockForUpdate()->first(); // Others wait until our transaction ends
    }

    public function recordStatus(Order $order, OrderStatus $to, ?User $actor, ?string $note): void
    {
        $from = $order->status;                       // Remember where it was
        $order->update(['status' => $to]);            // Move it
        $order->statusHistories()->create([           // Log it
            'user_id' => $actor?->id, 'from_status' => $from, 'to_status' => $to, 'note' => $note,
        ]);
    }

    public function expiredPendingReferences(\Carbon\CarbonInterface $before, int $limit): array
    {
        return Order::where('status', OrderStatus::Pending)
            ->where('created_at', '<', $before)
            ->orderBy('id')->limit($limit)   // Oldest first, in batches
            ->pluck('reference')->all();
    }
}

class StockRepository implements StockRepositoryInterface
{
    public function lockVariants(array $variants): Collection
    {
        if ($variants === []) {
            return new Collection();
        }

        return ProductStock::query()
            ->where(function ($q) use ($variants) {
                foreach ($variants as $v) {                                  // One OR-group per option
                    $q->orWhere(fn ($w) => $w
                        ->where('product_id', $v['product_id'])
                        ->where('color', $v['color'])
                        ->where('size', $v['size']));
                }
            })
            ->orderBy('product_id')->orderBy('color')->orderBy('size')      // Same lock order everywhere = no deadlocks
            ->lockForUpdate()                                                // Others wait until we commit
            ->get()
            ->keyBy(fn ($r) => ProductStock::key($r->product_id, $r->color, $r->size));
    }

    public function lockForProduct(Product $product): Collection
    {
        return ProductStock::where('product_id', $product->id)
            ->orderBy('color')->orderBy('size')
            ->lockForUpdate()
            ->get()
            ->keyBy(fn ($r) => ProductStock::key($r->product_id, $r->color, $r->size));
    }

    public function create(int $productId, string $color, string $size, int $quantity): ProductStock
    {
        return ProductStock::create(['product_id' => $productId, 'color' => $color, 'size' => $size, 'quantity' => $quantity]);
    }

    public function setQuantity(ProductStock $row, int $quantity): void
    {
        $row->update(['quantity' => $quantity]);
    }

    public function change(ProductStock $row, int $delta): void
    {
        $row->increment('quantity', $delta); // Done in SQL (quantity = quantity + delta), also updates the model
    }

    public function lowStock(int $threshold): Collection
    {
        return ProductStock::with('product:id,name,slug')
            ->where('quantity', '<=', $threshold)
            ->whereHas('product', fn ($q) => $q->where('is_published', true)) // Skip drafts and deleted products
            ->orderBy('quantity')->orderBy('product_id')
            ->get();
    }
}

<?php

namespace App\Services;

use App\DTOs\OrderFilterData;
use App\Enums\OrderStatus;
use App\Models\Order;
use App\Models\User;
use App\Repositories\Contracts\OrderRepositoryInterface;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;
use App\Events\OrderStatusChanged;


class AdminOrderService
{
    public function __construct(
        private OrderRepositoryInterface $orders,
        private StockService $stock,      // Gives stock back on cancellation
        private CatalogueCache $cache,
        ) {}

    public function list(OrderFilterData $filters): LengthAwarePaginator
    {
        return $this->orders->paginateForAdmin($filters);
    }

    public function find(string $reference): Order
    {
        return $this->orders->findForAdmin($reference) ?? throw new ModelNotFoundException(); // Becomes a 404
    }

    // Move an order to its next stage, safely even if two admins act at once
    public function changeStatus(User $actor, string $reference, OrderStatus $to, ?string $note): Order
    {
        DB::transaction(function () use ($actor, $reference, $to, $note) { // All or nothing
            $order = $this->orders->lockByReference($reference) ?? throw new ModelNotFoundException(); // Row is locked from here

            if (! $order->status->canMoveTo($to)) { // Checked against the FRESH status, after the lock
                throw ValidationException::withMessages([
                    'status' => ["An order that is {$order->status->label()} cannot be changed to {$to->label()}."],
                ]);
            }

            $this->orders->recordStatus($order, $to, $actor, $note);
            OrderStatusChanged::dispatch($order, $to); // Emails the customer for shipped / delivered / cancelled
            if ($to === OrderStatus::Cancelled) {
                $this->stock->release($order);                       // Items go back on the shelf (only if this order took stock)
                DB::afterCommit(fn () => $this->cache->flush());     // Public pages show them again
            }
        });

        return $this->find($reference); // Reload with history for the response
    }
}

use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;
use Symfony\Component\HttpKernel\Exception\ConflictHttpException;
use Throwable;
use App\Jobs\DeleteCloudinaryImage;

class AdminProductService
{
    public function __construct(
        private ProductRepositoryInterface $products,
        private ImageUploaderInterface $uploader, // Cloudinary
        private AuditLogger $audit,
        private CatalogueCache $cache,            // Public catalogue cache
    ) {}

    public function list(AdminProductFilterData $filters): LengthAwarePaginator
    {
        return $this->products->paginateForAdmin($filters);
    }

    public function find(int $id): Product
    {
        return $this->products->findForAdmin($id) ?? throw (new ModelNotFoundException)->setModel(Product::class, [$id]); // 404
    }

    public function create(User $actor, array $data): Product
    {
        $data = [...$this->defaults(), ...$data];                              // Fill anything the form left out
        $this->assertSoldOutSizes($data['sizes'], $data['sold_out_sizes']);    // Sold-out sizes must exist

        return DB::transaction(function () use ($actor, $data) {               // Product + log saved together
            $product = $this->products->create([...$data, 'slug' => $this->uniqueSlug($data['name'])]);
            $this->audit->record($actor, 'product.created', 'product', $product->id, "Created {$product->name}");
            $this->refreshCatalogue();

            return $this->find($product->id);
        });
    }

    // Edit part of a product. $version is the version the admin was looking at.
    public function update(User $actor, int $id, array $data, int $version): Product
    {
        return DB::transaction(function () use ($actor, $id, $data, $version) {
            $product = $this->products->lockById($id)                           // Row locked from here
                ?? throw (new ModelNotFoundException)->setModel(Product::class, [$id]);

            if ($product->version !== $version) {                               // Someone saved in between
                throw new ConflictHttpException('Someone else changed this product while you were editing. Reload it and try again.'); // 409
            }

            $this->assertSoldOutSizes($data['sizes'] ?? $product->sizes, $data['sold_out_sizes'] ?? $product->sold_out_sizes ?? []);

            $changes = $this->audit->diff($product->only(array_keys($data)), $data); // What really changed

            if ($changes === []) {
                return $this->find($id);                                        // Nothing changed: no new version, no log
            }

            $this->products->update($product, $data);
            $this->audit->record($actor, 'product.updated', 'product', $id, "Updated {$product->name}", $changes);
            $this->refreshCatalogue();

            return $this->find($id);
        });
    }

    public function delete(User $actor, int $id): void
    {
        DB::transaction(function () use ($actor, $id) {
            $product = $this->find($id);
            $this->products->delete($product);                                  // Hidden, not erased (photos stay too)
            $this->audit->record($actor, 'product.deleted', 'product', $id, "Deleted {$product->name}");
            $this->refreshCatalogue();
        });
    }

    // Upload a colour photo (pass $color) or a gallery photo (leave $color null)
    public function addPhoto(User $actor, int $id, UploadedFile $file, ?string $color, ?int $position): Product
    {
        $product = $this->find($id);

        if ($color !== null) {
            $match = collect($product->colors)->first(fn ($c) => strcasecmp($c['name'], $color) === 0); // Case-insensitive
            if (! $match) {
                throw ValidationException::withMessages(['color' => ["{$product->name} has no colour called {$color}."]]);
            }
            $colorSlug = Str::slug($match['name']);                             // "Dark Blue" -> dark-blue
            $position  = 1;                                                     // One photo per colour
            $publicId  = "{$product->slug}-{$colorSlug}";                       // Same naming as your existing photos
        } else {
            $colorSlug = null;
            $position  = $position ?? $this->products->nextGalleryPosition($product);
            if ($position > 6) {
                throw ValidationException::withMessages(['photo' => ['The gallery is full (6 photos). Delete one first.']]);
            }
            $publicId = $position === 1 ? $product->slug : "{$product->slug}-{$position}";
        }

        $uploaded = $this->uploader->upload($file->getRealPath(), $publicId);  // Send to Cloudinary first

        DB::transaction(function () use ($actor, $product, $colorSlug, $position, $uploaded) {
            $this->products->savePhoto($product, $colorSlug, $position, $uploaded['public_id'], $uploaded['url']);
            $label = $colorSlug ? "{$colorSlug} photo" : "gallery photo #{$position}";
            $this->audit->record($actor, 'product.photo_added', 'product', $product->id, "Added {$label} to {$product->name}");
            $this->refreshCatalogue();
        });

        return $this->find($id);
    }

    public function removePhoto(User $actor, int $id, int $photoId): Product
    {
        $product = $this->find($id);
        $photo = $this->products->findPhoto($product, $photoId)
            ?? throw (new ModelNotFoundException)->setModel(Product::class, [$photoId]); // 404

        DB::transaction(function () use ($actor, $product, $photo) {
            $this->products->deletePhoto($photo);
            $this->audit->record($actor, 'product.photo_removed', 'product', $product->id, "Removed a photo from {$product->name}");
            $this->refreshCatalogue();
        });

        try {
            dispatch(new DeleteCloudinaryImage($photo->public_id)); // Cloudinary cleanup runs in the background
        } catch (Throwable $e) {
            Log::warning('Could not delete Cloudinary image', ['public_id' => $photo->public_id, 'error' => $e->getMessage()]);
        }

        return $this->find($id);
    }

    // Values used when the form leaves a field out on create
    private function defaults(): array
    {
        return [
            'badge' => null, 'size_label' => SizeLabel::Size->value, 'features' => [], 'sold_out_sizes' => [],
            'is_featured' => false, 'is_sold_out' => false, 'is_published' => true,
        ];
    }

    // "XL" can't be sold out if the product has no XL
    private function assertSoldOutSizes(array $sizes, array $soldOut): void
    {
        $unknown = array_diff($soldOut, $sizes);

        if ($unknown) {
            throw ValidationException::withMessages([
                'sold_out_sizes' => ['These sold-out sizes are not in the sizes list: ' . implode(', ', $unknown) . '.'],
            ]);
        }
    }

    // Name -> slug, adding -2, -3... if it is taken
    private function uniqueSlug(string $name): string
    {
        $base = Str::slug($name);
        $slug = $base;
        $i = 2;

        while ($this->products->slugExists($slug)) {
            $slug = "{$base}-{$i}";
            $i++;
        }

        return $slug;
    }

    // Clear the public cache, but only once the database change is really saved
    private function refreshCatalogue(): void
    {
        DB::afterCommit(fn () => $this->cache->flush());
    }
}

<?php

namespace App\Services;

use App\Models\Product;
use App\Models\ProductStock;
use App\Models\User;
use App\Repositories\Contracts\ProductRepositoryInterface;
use App\Repositories\Contracts\StockRepositoryInterface;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class AdminStockService
{
    public function __construct(
        private ProductRepositoryInterface $products,
        private StockRepositoryInterface $stock,
        private AuditLogger $audit,
        private CatalogueCache $cache,
    ) {}

    public function show(int $productId): Product
    {
        return $this->products->findForAdmin($productId) ?? throw (new ModelNotFoundException)->setModel(Product::class, [$productId]);
    }

    public function lowStock(): Collection
    {
        return $this->stock->lowStock((int) config('shop.low_stock_threshold'));
    }

    // $entries: [['color' => 'Black', 'size' => 'M', 'set' => 10] or ['color' => ..., 'size' => ..., 'add' => 5], ...]
    public function adjust(User $actor, int $productId, array $entries): Product
    {
        $product = $this->show($productId);
        $lines = [];

        foreach ($entries as $i => $entry) {                       // Match every entry to a real option (canonical spelling)
            $color = collect($product->colors)->first(fn ($c) => strcasecmp($c['name'], $entry['color']) === 0)['name'] ?? null;
            $size  = collect($product->sizes)->first(fn ($s) => strcasecmp($s, $entry['size']) === 0);

            if ($color === null || $size === null) {
                throw ValidationException::withMessages(["stock.{$i}" => ["{$product->name} has no option {$entry['color']} / {$entry['size']}."]]);
            }

            $lines[] = ['color' => $color, 'size' => $size, 'set' => $entry['set'] ?? null, 'add' => $entry['add'] ?? null];
        }

        DB::transaction(function () use ($actor, $product, $lines) {
            $rows = $this->stock->lockForProduct($product);        // Lock all of this product's rows
            $changes = [];

            foreach ($lines as $i => $line) {
                $key    = ProductStock::key($product->id, $line['color'], $line['size']);
                $row    = $rows->get($key);
                $before = $row?->quantity ?? 0;
                $after  = $line['set'] !== null ? (int) $line['set'] : $before + (int) $line['add']; // Absolute or relative

                if ($after < 0) {
                    throw ValidationException::withMessages(["stock.{$i}" => ["Stock cannot go below 0 ({$line['color']} / {$line['size']} has {$before})."]]);
                }

                if ($row) {
                    $this->stock->setQuantity($row, $after);
                } else {
                    $rows->put($key, $this->stock->create($product->id, $line['color'], $line['size'], $after));
                }

                if ($before !== $after) {
                    $changes["{$line['color']} / {$line['size']}"] = ['from' => $before, 'to' => $after];
                }
            }

            if ($changes !== []) {
                $this->audit->record($actor, 'stock.updated', 'product', $product->id, "Updated stock for {$product->name}", $changes);
                DB::afterCommit(fn () => $this->cache->flush());   // Public pages show the new availability
            }
        });

        return $this->show($productId);                            // Reload with fresh numbers
    }
}

<?php

namespace App\Services;

use App\Enums\OrderStatus;
use App\Repositories\Contracts\AnalyticsRepositoryInterface;
use App\Repositories\Contracts\StockRepositoryInterface;
use Carbon\Carbon;
use Closure;
use Illuminate\Support\Facades\Cache;

class AnalyticsService
{
    public function __construct(
        private AnalyticsRepositoryInterface $analytics,
        private StockRepositoryInterface $stock,
    ) {}

    public function summary(int $days): array
    {
        return $this->cached("summary:{$days}", function () use ($days) {
            $from = $this->windowStart($days);
            $now  = now();
            $cur  = $this->analytics->totals($from, $now);
            $prev = $this->analytics->totals($from->copy()->subDays($days), $from); // Same length, just before

            $counts = $this->analytics->statusCounts();
            $byStatus = [];
            foreach (OrderStatus::cases() as $status) {
                $byStatus[$status->value] = (int) ($counts[$status->value] ?? 0);  // Zero for empty statuses
            }

            return [
                'days'              => $days,
                'revenue'           => $cur['revenue'],
                'orders'            => $cur['orders'],
                'averageOrderValue' => $cur['orders'] > 0 ? (int) round($cur['revenue'] / $cur['orders']) : 0,
                'previous'          => $prev,
                'change'            => [
                    'revenuePercent' => $this->percent($cur['revenue'], $prev['revenue']),
                    'ordersPercent'  => $this->percent($cur['orders'], $prev['orders']),
                ],
                'ordersByStatus'    => $byStatus,
                'lowStockOptions'   => $this->stock->lowStock((int) config('shop.low_stock_threshold'))->count(),
            ];
        });
    }

    public function revenue(int $days): array
    {
        return $this->cached("revenue:{$days}", function () use ($days) {
            $tz = config('shop.timezone');
            $offset = now($tz)->utcOffset() - now(config('app.timezone'))->utcOffset(); // Minutes between Lagos and the app clock
            $rows = $this->analytics->daily($this->windowStart($days), now(), $offset);

            $series = [];
            for ($i = $days - 1; $i >= 0; $i--) {                                    // Every day appears, even with no sales
                $day = now($tz)->subDays($i)->toDateString();
                $series[] = [
                    'date'    => $day,
                    'revenue' => (int) ($rows[$day]->revenue ?? 0),
                    'orders'  => (int) ($rows[$day]->orders ?? 0),
                ];
            }

            return $series;
        });
    }

    public function topProducts(int $days, int $limit): array
    {
        return $this->cached("top:{$days}:{$limit}", function () use ($days, $limit) {
            return array_map(fn ($r) => [
                'productId' => (int) $r->product_id,
                'name'      => $r->name,
                'units'     => (int) $r->units,
                'revenue'   => (int) $r->revenue,
            ], $this->analytics->topProducts($this->windowStart($days), now(), $limit));
        });
    }

    // Start of the day ($days - 1) ago in Lagos, expressed in the app's timezone (what the database stores)
    private function windowStart(int $days): Carbon
    {
        return now(config('shop.timezone'))->subDays($days - 1)->startOfDay()->setTimezone(config('app.timezone'));
    }

    private function percent(int $current, int $previous): ?float
    {
        return $previous === 0 ? null : round(($current - $previous) / $previous * 100, 1); // null = nothing to compare with
    }

    // Short cache: dashboards refresh often and one minute of delay is fine
    private function cached(string $key, Closure $callback): mixed
    {
        $ttl = (int) config('shop.analytics_cache_ttl');

        return $ttl > 0 ? Cache::remember("analytics:{$key}", $ttl, $callback) : $callback();
    }
}

<?php

namespace App\Services;

use App\Models\AuditLog;
use App\Models\User;
use Illuminate\Pagination\LengthAwarePaginator;

class AuditLogger
{
    // Write one log entry
    public function record(User $actor, string $action, string $subjectType, int $subjectId, string $summary, array $changes = []): void
    {
        AuditLog::create([
            'user_id'      => $actor->id,
            'action'       => $action,
            'subject_type' => $subjectType,
            'subject_id'   => $subjectId,
            'summary'      => $summary,
            'changes'      => $changes ?: null, // Store null instead of an empty list
            'ip_address'   => request()->ip(),
        ]);
    }

    // Compare before/after and keep only the fields that really changed
    public function diff(array $before, array $after): array
    {
        $changes = [];

        foreach ($after as $key => $value) {
            $old = $before[$key] ?? null;

            if (json_encode($old) !== json_encode($value)) { // JSON compare handles arrays and enums
                $changes[$key] = ['from' => $old, 'to' => $value];
            }
        }

        return $changes;
    }

    // Newest first, with optional filters (used by the super admin log screen)
    public function list(?string $action, ?string $subjectType, int $perPage = 30): LengthAwarePaginator
    {
        return AuditLog::with('user')
            ->when($action, fn ($q, $a) => $q->where('action', $a))
            ->when($subjectType, fn ($q, $t) => $q->where('subject_type', $t))
            ->latest('id')
            ->paginate($perPage);
    }
}

<?php

namespace App\Services;

use App\DTOs\LoginData;
use App\DTOs\RegisterData;
use App\Models\User;
use App\Repositories\Contracts\UserRepositoryInterface;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class AuthService
{
    public function __construct(private UserRepositoryInterface $users) {} // Injected by the container

    // Create the account and log it in straight away
    public function register(RegisterData $data): array
    {
        $user = $this->users->create($data);

        return ['user' => $user, 'token' => $this->issueToken($user)];
    }

    // Check the credentials and hand back a fresh token
    public function login(LoginData $data): array
    {
        $user = $this->users->findByEmail($data->email);

        if (! $user || ! Hash::check($data->password, $user->password)) {
            throw ValidationException::withMessages([
                'email' => ['These credentials do not match our records.'], // Same message for both cases
            ]);
        }

        return ['user' => $user, 'token' => $this->issueToken($user)];
    }

    // Revoke only the token used for this request (other devices stay logged in)
    public function logout(User $user): void
    {
        $user->currentAccessToken()->delete();
    }

    private function issueToken(User $user): string
    {
        return $user->createToken('web')->plainTextToken; // The raw token is shown only once
    }
}

<?php

namespace App\Services;

use Closure;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Str;

class CatalogueCache
{
    // Return the cached value, or run $callback, store its result and return it
    public function remember(string $key, Closure $callback): mixed
    {
        $ttl = (int) config('shop.catalogue_cache_ttl');

        if ($ttl <= 0) {
            return $callback(); // Caching switched off (handy while debugging)
        }

        return Cache::remember("catalogue:{$this->version()}:{$key}", $ttl, $callback);
    }

    // Admin changed something: a new version makes every old entry unreachable
    public function flush(): void
    {
        Cache::forever('catalogue:version', (string) Str::uuid());
    }

    // The current version token (created on first use)
    private function version(): string
    {
        return Cache::rememberForever('catalogue:version', fn () => (string) Str::uuid());
    }
}

<?php

namespace App\Services;

use App\Enums\OrderStatus;
use App\Repositories\Contracts\OrderRepositoryInterface;
use Illuminate\Support\Facades\DB;
use App\Events\OrderStatusChanged;

class OrderExpiryService
{
    public function __construct(
        private OrderRepositoryInterface $orders,
        private StockService $stock,
        private CatalogueCache $cache,
    ) {}

    // Cancel unpaid orders older than $minutes. Returns how many were cancelled.
    public function expire(int $minutes): int
    {
        $references = $this->orders->expiredPendingReferences(now()->subMinutes($minutes), 200); // Batch of 200 per run
        $cancelled = 0;

        foreach ($references as $reference) {
            $done = DB::transaction(function () use ($reference, $minutes) {
                $order = $this->orders->lockByReference($reference);   // Lock the row, then re-check

                if (! $order || $order->status !== OrderStatus::Pending) {
                    return false;                                      // Paid or changed since we looked: leave it
                }

                $this->orders->recordStatus($order, OrderStatus::Cancelled, null, "Expired: not paid within {$minutes} minutes"); // null = System
                $this->stock->release($order);                         // Stock back on the shelf

                OrderStatusChanged::dispatch($order, OrderStatus::Cancelled); // Tell the customer

                return true;
            }, 3);

            $cancelled += $done ? 1 : 0;
        }

        if ($cancelled > 0) {
            $this->cache->flush();                                     // Public pages show the stock again
        }

        return $cancelled;
    }
}


===== .\app\Services\OrderService.php =====
<?php

namespace App\Services;

use App\DTOs\CartLineData;
use App\DTOs\CheckoutData;
use App\Enums\OrderStatus;
use App\Models\Order;
use App\Models\Product;
use App\Models\User;
use App\Repositories\Contracts\OrderRepositoryInterface;
use App\Repositories\Contracts\ProductRepositoryInterface;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Database\UniqueConstraintViolationException;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;
use Symfony\Component\HttpKernel\Exception\TooManyRequestsHttpException;

class OrderService
{
    public function __construct(
        private OrderRepositoryInterface $orders,
        private ProductRepositoryInterface $products,
        private StockService $stock,       // Reserves stock under a lock
        private CatalogueCache $cache,     // Refreshed when stock changes
    ) {}

    // Turn a cart into an order. Every price and option is checked against the database.
    public function place(?User $user, CheckoutData $data, ?string $idempotencyKey = null): Order
    {
        // Same attempt seen before? Return the order we already made.
        if ($idempotencyKey && $existing = $this->orders->findByIdempotencyKey($user, $data->email, $idempotencyKey)) {
            return $existing;
        }

        // Cap unpaid orders per email so nobody can hoard stock without paying
        if ($this->orders->countPendingForEmail($data->email) >= config('shop.max_pending_per_email', 3)) {
            throw new TooManyRequestsHttpException(null, 'You have unpaid orders waiting. Please pay for them, or wait for them to expire, before placing another.');
        }

        $found = $this->products->findManyByIds(
            collect($data->lines)->pluck('productId')->unique()->values()->all()
        );

        $items = [];
        $subtotal = 0;

        foreach ($data->lines as $index => $line) {
            $product = $found->get($line->productId);
            $this->assertBuyable($product, $line, $index);

            $items[] = [
                'product_id' => $product->id,
                'name'       => $product->name,                          // Copied, so history never changes
                'color'      => $line->color,
                'size'       => $line->size,
                'image'      => $this->photoFor($product, $line->color), // That colour's photo, if any
                'unit_price' => $product->price,                         // From the database, never from the browser
                'quantity'   => $line->quantity,
            ];
            $subtotal += $product->price * $line->quantity;
        }

        $deliveryFee = $this->deliveryFee($subtotal, $data->state);

        try {
            return DB::transaction(function () use ($user, $data, $items, $subtotal, $deliveryFee, $idempotencyKey) {
                $this->stock->reserve($items); // Lock, check and take the stock (throws a 422 if short)

                $order = $this->orders->create($user, [
                    'email'             => $data->email,          // Receipts and lookups use this
                    'status'            => OrderStatus::Pending,
                    'subtotal'          => $subtotal,
                    'delivery_fee'      => $deliveryFee,
                    'total'             => $subtotal + $deliveryFee,
                    'recipient_name'    => $data->recipientName,
                    'phone'             => $data->phone,
                    'address'           => $data->address,
                    'city'              => $data->city,
                    'state'             => $data->state,
                    'idempotency_key'   => $idempotencyKey,   // Saved so repeats can be recognised
                    'stock_reserved_at' => now(),             // This order holds stock until it is cancelled
                ], $items);

                DB::afterCommit(fn () => $this->cache->flush()); // Availability changed: refresh public pages

                return $order;
            }, 3); // Retry up to 3 times if MySQL reports a deadlock
        } catch (UniqueConstraintViolationException $e) {
            // Two identical requests raced: the other one won, so return its order
            return ($idempotencyKey ? $this->orders->findByIdempotencyKey($user, $data->email, $idempotencyKey) : null) ?? throw $e;
        }
    }

    public function listFor(User $user): Collection
    {
        return $this->orders->forUser($user);
    }

    // Account owner, or a guest who knows the order's email
    public function findFor(?User $user, string $reference, ?string $email = null): Order
    {
        return $this->orders->findForCustomer($user, $email, $reference) ?? throw new ModelNotFoundException(); // Becomes a 404
    }

    // Free delivery over the threshold, otherwise a flat fee by state
    private function deliveryFee(int $subtotal, string $state): int
    {
        if ($subtotal >= config('shop.free_delivery_over')) {
            return 0;
        }

        return strcasecmp(trim($state), 'Lagos') === 0
            ? config('shop.delivery_fee_lagos')
            : config('shop.delivery_fee_other');
    }

    // Reject lines that don't match a real, available product option
    private function assertBuyable(?Product $product, CartLineData $line, int $index): void
    {
        $problem = match (true) {
            $product === null || $product->is_sold_out || ! $product->is_published
                => 'This product is no longer available.',
            ! collect($product->colors ?? [])->contains('name', $line->color)
                => "{$product->name} is not available in {$line->color}.",
            ! in_array($line->size, $product->sizes ?? [], true)
                => "{$product->name} does not come in {$line->size}.",
            in_array($line->size, $product->sold_out_sizes ?? [], true)
                => "{$product->name} in {$line->size} is sold out.",
            default => null,
        };

        if ($problem) {
            throw ValidationException::withMessages(["items.$index" => [$problem]]); // 422 pointing at the bad line
        }
    }

    // The Cloudinary photo for the chosen colour, or null
    private function photoFor(Product $product, string $color): ?string
    {
        return $product->photos->firstWhere('color', Str::slug($color))?->url;
    }
}

<?php

namespace App\Services;

use App\Enums\OrderStatus;
use App\Models\Order;
use App\Models\User;
use App\Events\OrderPaid;
use App\Repositories\Contracts\OrderRepositoryInterface;
use App\Services\Contracts\PaymentGatewayInterface;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class PaymentService
{
    public function __construct(
        private PaymentGatewayInterface $gateway,  // Paystack (bound in the provider)
        private OrderRepositoryInterface $orders,
    ) {}

    // Start paying for a pending order (mine, or a guest order with the right email); returns the Paystack page URL
    public function start(?User $user, string $reference, ?string $email): string
    {
        $order = $this->orders->findForCustomer($user, $email, $reference) ?? throw new ModelNotFoundException(); // 404 if not theirs

        if ($order->status !== OrderStatus::Pending) {
            throw ValidationException::withMessages(['order' => ['This order is not awaiting payment.']]);
        }

        $paymentReference = $order->reference . '-' . strtoupper(Str::random(4)); // New id per attempt (Paystack rejects repeats)

        $url = $this->gateway->initialize(
            $paymentReference,
            $order->total * 100,                                       // Naira to kobo
            $order->email,                                             // The order's own email (works for guests)
            config('shop.frontend_url') . "/order/{$order->reference}", // Customer returns here
            ['order_reference' => $order->reference],
        );

        $this->orders->attachPaymentReference($order, $paymentReference); // Remember which attempt is live

        return $url;
    }

    // Customer came back: check with Paystack and update the order if paid
    public function confirm(?User $user, string $reference, ?string $email): Order
    {
        $order = $this->orders->findForCustomer($user, $email, $reference) ?? throw new ModelNotFoundException();

        if ($order->status === OrderStatus::Pending && $order->payment_reference) {
            $this->settle($order, $order->payment_reference);
        }

        return $this->orders->findForCustomer($user, $email, $reference); // Reload with the new status
    }

    // Paystack called us: verify the signature, then settle the order
    public function handleWebhook(string $payload, ?string $signature): void
    {
        $secret = (string) config('services.paystack.secret_key');

        if ($secret === '') {
            abort(500, 'Payments are not configured.'); // Never accept webhooks with no key
        }

        $expected = hash_hmac('sha512', $payload, $secret); // What a genuine signature looks like

        if (! $signature || ! hash_equals($expected, $signature)) {
            abort(401, 'Invalid signature.'); // Not from Paystack
        }

        $event = json_decode($payload, true);

        if (($event['event'] ?? null) !== 'charge.success') {
            return; // We only care about successful charges
        }

        $paymentReference = $event['data']['reference'] ?? null;
        $orderReference   = $event['data']['metadata']['order_reference'] ?? null;
        $order = $orderReference ? $this->orders->findByReference($orderReference) : null;

        if ($order && $paymentReference && $order->status === OrderStatus::Pending) {
            $this->settle($order, $paymentReference); // Re-verify; never trust the payload alone
        }
    }

    // Ask Paystack, check the amount, then mark the order paid
    private function settle(Order $order, string $paymentReference): void
    {
        $result = $this->gateway->verify($paymentReference);

        if (! $result->paid) {
            return; // Not paid (yet)
        }

        if ($result->amount !== $order->total * 100 || $result->currency !== 'NGN') {
            Log::warning('Payment amount mismatch', ['order' => $order->reference]); // Suspicious: do not mark paid
            return;
        }

        if ($this->orders->markPaid($order, $paymentReference)) { // True only for the call that really flipped it
            OrderPaid::dispatch($order);                           // So the emails go out exactly once
        }
    }
}

<?php

namespace App\Services;

use App\DTOs\ProductFilterData;
use App\Http\Resources\ProductResource;
use App\Models\Product;
use App\Repositories\Contracts\ProductRepositoryInterface;
use Illuminate\Database\Eloquent\ModelNotFoundException;

class ProductService
{
    public function __construct(
        private ProductRepositoryInterface $products, // interface, not the Eloquent class
        private CatalogueCache $cache,                // our versioned cache
    ) {}

    // Returns the finished JSON-ready list (small to cache), not Eloquent models (huge to cache)
    public function list(ProductFilterData $filters): array
    {
        $build = fn () => ProductResource::collection($this->products->getFiltered($filters))->resolve();

        if ($filters->search !== null) {
            return $build(); // Never cache free-text searches (cache flooding)
        }

        $key = 'products:v2:' . md5(json_encode([$filters->category, $filters->featured, $filters->sort?->value]));

        return $this->cache->remember($key, $build);
    }

    // One product as JSON-ready data
    public function findBySlug(string $slug): array
    {
        return $this->cache->remember("product:v2:{$slug}", function () use ($slug) {
            $product = $this->products->findBySlug($slug)
                ?? throw (new ModelNotFoundException)->setModel(Product::class, [$slug]); // A 404 is never cached

            return (new ProductResource($product))->resolve();
        });
    }
}

<?php

namespace App\Services;

use App\Enums\UserRole;
use App\Models\User;
use App\Repositories\Contracts\StaffRepositoryInterface;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Database\Eloquent\ModelNotFoundException;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class StaffService
{
    public function __construct(
        private StaffRepositoryInterface $staff,
        private AuditLogger $audit,
    ) {}

    public function list(): Collection
    {
        return $this->staff->list();
    }

    // New staff account, or promote an existing customer who has this email
    public function create(User $actor, array $data): User
    {
        $role = UserRole::from($data['role']);

        return DB::transaction(function () use ($actor, $data, $role) {
            $existing = $this->staff->findByEmail($data['email']);

            if ($existing) {
                if ($existing->role !== UserRole::Customer) {
                    throw ValidationException::withMessages(['email' => ['This person is already on the staff.']]);
                }

                $this->staff->update($existing, ['role' => $role, 'is_active' => true]); // Customer -> staff
                $this->audit->record($actor, 'staff.promoted', 'user', $existing->id, "Promoted {$existing->email} to {$role->value}");

                return $existing->refresh();
            }

            if (empty($data['name']) || empty($data['password'])) {                      // Brand-new account needs both
                throw ValidationException::withMessages(['name' => ['Name and password are required for a new staff account.']]);
            }

            $user = $this->staff->create([
                'name'      => $data['name'],
                'email'     => $data['email'],
                'password'  => Hash::make($data['password']),
                'role'      => $role,
                'is_active' => true,
            ]);

            $this->audit->record($actor, 'staff.created', 'user', $user->id, "Created {$role->value} {$user->email}");

            return $user;
        });
    }

    // Change role and/or active status
    public function update(User $actor, int $id, array $changes): User
    {
        return DB::transaction(function () use ($actor, $id, $changes) {
            $superAdmins = $this->staff->lockActiveSuperAdmins();                       // Always lock these first
            $user = $this->staff->lockStaffById($id)
                ?? throw (new ModelNotFoundException)->setModel(User::class, [$id]);    // 404 (also for customers)

            if ($user->is($actor)) {
                throw ValidationException::withMessages(['user' => ['You cannot change your own role or status.']]);
            }

            $newRole   = isset($changes['role']) ? UserRole::from($changes['role']) : $user->role;
            $newActive = $changes['is_active'] ?? (bool) $user->is_active;

            // Would this change remove an active super admin from the pool?
            $losesSuper = $user->role === UserRole::SuperAdmin && $user->is_active
                && ($newRole !== UserRole::SuperAdmin || ! $newActive);

            if ($losesSuper && $superAdmins->reject(fn ($u) => $u->id === $user->id)->isEmpty()) {
                throw ValidationException::withMessages(['user' => ['There must always be at least one active super admin.']]);
            }

            $before = ['role' => $user->role->value, 'is_active' => (bool) $user->is_active];
            $after  = ['role' => $newRole->value,    'is_active' => (bool) $newActive];
            $diff   = $this->audit->diff($before, $after);

            if ($diff === []) {
                return $user;                                                           // Nothing changed
            }

            $this->staff->update($user, ['role' => $newRole, 'is_active' => $newActive]);

            if (! $newActive) {
                $this->staff->revokeTokens($user);                                      // Deactivated = logged out now
            }

            $this->audit->record($actor, 'staff.updated', 'user', $user->id, "Updated {$user->email}", $diff);

            return $user->refresh();
        }, 3); // Retry if MySQL reports a deadlock
    }
}

<?php

namespace App\Services;

use App\Models\Order;
use App\Models\ProductStock;
use App\Repositories\Contracts\StockRepositoryInterface;
use Illuminate\Validation\ValidationException;

class StockService
{
    public function __construct(private StockRepositoryInterface $stock) {}

    // Take stock for an order's lines. MUST run inside a DB transaction (the locks last until it ends).
    public function reserve(array $items): void
    {
        $needs = []; // key => total wanted for that option (the same option can appear twice)

        foreach ($items as $index => $item) {
            $key = ProductStock::key($item['product_id'], $item['color'], $item['size']);
            $needs[$key] ??= [
                'product_id' => $item['product_id'], 'color' => $item['color'], 'size' => $item['size'],
                'name' => $item['name'], 'qty' => 0, 'index' => $index,
            ];
            $needs[$key]['qty'] += $item['quantity'];
        }

        $rows = $this->stock->lockVariants(array_values($needs)); // Lock first, THEN read: nobody can sneak in between

        foreach ($needs as $key => $need) {                       // Check everything before changing anything
            $left = $rows->get($key)?->quantity ?? 0;

            if ($left < $need['qty']) {
                $label = "{$need['name']} ({$need['color']}, {$need['size']})";
                throw ValidationException::withMessages([
                    "items.{$need['index']}" => [$left === 0 ? "{$label} is sold out." : "Only {$left} left of {$label}."],
                ]);
            }
        }

        foreach ($needs as $key => $need) {
            $this->stock->change($rows->get($key), -$need['qty']); // Take it off the shelf
        }
    }

    // Put an order's stock back (cancelled). MUST run inside a DB transaction.
    public function release(Order $order): void
    {
        if ($order->stock_reserved_at === null) {
            return; // This order never took stock (placed before stock tracking, or already released)
        }

        $totals = [];

        foreach ($order->items()->get() as $item) {
            if ($item->product_id === null) {
                continue;
            }
            $key = ProductStock::key($item->product_id, $item->color, $item->size);
            $totals[$key] ??= ['product_id' => $item->product_id, 'color' => $item->color, 'size' => $item->size, 'qty' => 0];
            $totals[$key]['qty'] += $item->quantity;
        }

        $rows = $this->stock->lockVariants(array_values($totals));

        foreach ($totals as $key => $t) {
            if ($row = $rows->get($key)) {
                $this->stock->change($row, $t['qty']); // Back on the shelf
            }
        }

        $order->forceFill(['stock_reserved_at' => null])->save(); // Never release the same order twice
    }
}

class CloudinaryImageUploader implements ImageUploaderInterface
{
    private Cloudinary $client; // SDK client, built once per instance

    {
        $this->client = new Cloudinary([
            'cloud' => [
                'cloud_name' => config('cloudinary.cloud_name'), // Which account
                'api_key'    => config('cloudinary.api_key'),    // Who we are
                'api_secret' => config('cloudinary.api_secret'), // Proof
            ],
            'url' => ['secure' => true], // Always return https URLs
        ]);
    }

    {
        if (! is_file($path)) {
            throw new RuntimeException("Image not found: {$path}"); // Fail early on a bad path
        }

            $result = $this->client->uploadApi()->upload($path, [
                'public_id'     => $publicId,                    // File name inside Cloudinary
                'folder'        => config('cloudinary.folder'),  // Keeps products together
                'overwrite'     => true,                         // Re-running replaces, no duplicates
                'resource_type' => 'image',
            ]);
        } catch (Throwable $e) {
            throw new RuntimeException('Cloudinary upload failed: ' . $e->getMessage(), 0, $e);
        }

        return [
            'public_id' => $result['public_id'],  // Used later to delete or transform
            'url'       => $result['secure_url'], // Full https link to the image
        ];
    }

        public function delete(string $publicId): void
    {
        try {
            $this->client->uploadApi()->destroy($publicId); // Remove the file from Cloudinary
        } catch (Throwable $e) {
            throw new RuntimeException('Cloudinary delete failed: ' . $e->getMessage(), 0, $e);
        }
    }
}


===== .\app\Services\Payments\PaystackGateway.php =====
<?php

namespace App\Services\Payments;

use App\DTOs\PaymentVerification;
use App\Exceptions\PaymentException;
use App\Services\Contracts\PaymentGatewayInterface;
use Illuminate\Http\Client\PendingRequest;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class PaystackGateway implements PaymentGatewayInterface
{
    // Shared request setup: base URL, secret key, JSON, timeout
    private function http(): PendingRequest
    {
        return Http::baseUrl(config('services.paystack.base_url'))
            ->withToken((string) config('services.paystack.secret_key'))
            ->acceptJson()
            ->timeout(15);
    }

    public function initialize(string $reference, int $amountKobo, string $email, string $callbackUrl, array $metadata = []): string
    {
        $res = $this->http()->post('/transaction/initialize', [
            'reference'    => $reference,    // Our unique id for this attempt
            'amount'       => $amountKobo,   // Paystack wants kobo (naira x 100)
            'email'        => $email,
            'currency'     => 'NGN',
            'callback_url' => $callbackUrl,  // Where the customer lands after paying
            'metadata'     => $metadata,     // Lets the webhook find our order
        ]);

        if ($res->failed() || ! $res->json('status')) {
            Log::error('Paystack initialize failed', ['body' => $res->body()]); // Details go to the log only
            throw new PaymentException('We could not start the payment. Please try again.');
        }

        return $res->json('data.authorization_url');
    }

    public function verify(string $reference): PaymentVerification
    {
        $res = $this->http()->get('/transaction/verify/' . rawurlencode($reference));

        if ($res->failed()) {
            Log::error('Paystack verify failed', ['body' => $res->body()]);
            throw new PaymentException('We could not confirm the payment yet. Please try again.');
        }

        return new PaymentVerification(
            paid: $res->json('data.status') === 'success', // Only "success" counts
            amount: (int) $res->json('data.amount', 0),
            currency: (string) $res->json('data.currency', ''),
        );
    }
}

===== routes\api.php =====
<?php

use App\Http\Controllers\Api\V1\CategoryController;
use App\Http\Controllers\Api\V1\ProductController;
use App\Http\Controllers\Api\V1\AuthController;
use App\Http\Controllers\Api\V1\OrderController;
use App\Http\Controllers\Api\V1\PaymentController;
use App\Http\Controllers\Api\V1\Admin\OrderController as AdminOrderController;
use App\Http\Controllers\Api\V1\Admin\AuditLogController;
use App\Http\Controllers\Api\V1\Admin\ProductController as AdminProductController;
use App\Http\Controllers\Api\V1\Admin\StockController as AdminStockController;
use App\Http\Controllers\Api\V1\Admin\AnalyticsController as AdminAnalyticsController;
use App\Http\Controllers\Api\V1\Admin\StaffController as AdminStaffController;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Route;
use App\Http\Resources\UserResource;

// Default Sanctum route: returns the logged-in user (used later for accounts)
Route::get('/user', fn (Request $request) => $request->user())->middleware('auth:sanctum');

// Version 1 of the public API: URLs look like /api/v1/products
Route::prefix('v1')->group(function () {
    // Public catalogue: browsers may reuse a response for 60s, and answer "304 Not Modified" via the ETag
    Route::middleware('cache.headers:public;max_age=60;etag')->group(function () {
        Route::get('/categories', [CategoryController::class, 'index']);
        Route::get('/products', [ProductController::class, 'index']);
        Route::get('/products/{slug}', [ProductController::class, 'show']);
    });
    Route::post('/webhooks/paystack', [PaymentController::class, 'webhook']); // Paystack calls this itself

    Route::post('/register', [AuthController::class, 'register'])->middleware('throttle:10,1'); // 10 tries a minute
    Route::post('/login', [AuthController::class, 'login'])->middleware('throttle:5,1');        // 5 tries a minute

    // Everything under /api/v1/admin needs a login AND a staff role
    Route::prefix('admin')->middleware(['auth:sanctum', 'staff'])->group(function () {
        Route::get('/me', fn (Request $request) => new UserResource($request->user())); // Who am I (staff view)

        Route::get('/orders', [AdminOrderController::class, 'index']);                             // List + filter
        Route::get('/orders/{reference}', [AdminOrderController::class, 'show']);                  // One order
        Route::patch('/orders/{reference}/status', [AdminOrderController::class, 'updateStatus']); // Change stage

        Route::get('/products', [AdminProductController::class, 'index']);                          // List (drafts included)
        Route::post('/products', [AdminProductController::class, 'store']);                         // Create
        Route::get('/products/{id}', [AdminProductController::class, 'show'])->whereNumber('id');
        Route::patch('/products/{id}', [AdminProductController::class, 'update'])->whereNumber('id');   // Edit
        Route::delete('/products/{id}', [AdminProductController::class, 'destroy'])->whereNumber('id'); // Soft delete
        Route::post('/products/{id}/photos', [AdminProductController::class, 'storePhoto'])->whereNumber('id');
        Route::delete('/products/{id}/photos/{photoId}', [AdminProductController::class, 'destroyPhoto'])->whereNumber(['id', 'photoId']);

        Route::get('/audit-logs', [AuditLogController::class, 'index'])->middleware('can:manage-staff'); // Super admin only

        Route::get('/products/{id}/stock', [AdminStockController::class, 'show'])->whereNumber('id');    // Stock matrix
        Route::patch('/products/{id}/stock', [AdminStockController::class, 'update'])->whereNumber('id'); // Set / add stock
        Route::get('/stock/low', [AdminStockController::class, 'low']);                                    // Running low

        Route::prefix('analytics')->group(function () {
            Route::get('/summary', [AdminAnalyticsController::class, 'summary']);          // Totals + comparison
            Route::get('/revenue', [AdminAnalyticsController::class, 'revenue']);          // Daily chart data
            Route::get('/top-products', [AdminAnalyticsController::class, 'topProducts']); // Best sellers
        });

        Route::middleware('can:manage-staff')->prefix('staff')->group(function () {              // Super admin only
            Route::get('/', [AdminStaffController::class, 'index']);                             // List staff
            Route::post('/', [AdminStaffController::class, 'store']);                            // Create / promote
            Route::patch('/{id}', [AdminStaffController::class, 'update'])->whereNumber('id');   // Role / active
        });
    });

    // Guests and signed-in customers share these (guests prove ownership with the order's email)
    Route::post('/orders', [OrderController::class, 'store'])->middleware('throttle:6,1');                          // Place an order
    Route::post('/orders/{reference}/lookup', [OrderController::class, 'lookup'])->middleware('throttle:15,1');     // Track an order
    Route::post('/orders/{reference}/pay', [PaymentController::class, 'pay'])->middleware('throttle:10,1');         // Start payment
    Route::post('/orders/{reference}/confirm', [PaymentController::class, 'confirm'])->middleware('throttle:20,1'); // Check payment

    // Account-only routes: a valid token is required
    Route::middleware('auth:sanctum')->group(function () {
        Route::get('/me', [AuthController::class, 'me']);
        Route::post('/logout', [AuthController::class, 'logout']);

        Route::get('/orders', [OrderController::class, 'index']);                // My orders
        Route::get('/orders/{reference}', [OrderController::class, 'show']);     // One of my orders
    });
});
