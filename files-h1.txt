"use client"; // uses state and browser storage, so it runs in the browser

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";
import {
  ApiError,
  loginRequest,
  logoutRequest,
  meRequest,
  registerRequest,
  type AuthResult,
  type User,
} from "@/lib/authApi";

// Everything other components can read or call
type AuthContextValue = {
  user: User | null;  // null = signed out
  ready: boolean;     // true once we've checked for a saved login (avoids a "signed out" flash)
  register: (input: { name: string; email: string; phone?: string; password: string }) => Promise<void>;
  login: (email: string, password: string) => Promise<void>;
  logout: () => Promise<void>;
};

const AuthContext = createContext<AuthContextValue | null>(null);

const TOKEN_KEY = "be-tender-token"; // where the login token is kept in the browser

// Small helpers: storage can be blocked, so every access is wrapped
const readToken = (): string | null => {
  try {
    return localStorage.getItem(TOKEN_KEY);
  } catch {
    return null;
  }
};
const writeToken = (token: string | null) => {
  try {
    if (token) localStorage.setItem(TOKEN_KEY, token);
    else localStorage.removeItem(TOKEN_KEY);
  } catch {
    /* storage blocked: the login just won't survive a refresh */
  }
};

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [ready, setReady] = useState(false);

  // On first load: if a token was saved, ask the API who it belongs to
  useEffect(() => {
    const token = readToken();
    if (!token) {
      setReady(true);
      return;
    }
    meRequest(token)
      .then((r) => setUser(r.user))
      .catch((e) => {
        if (e instanceof ApiError && e.status === 401) writeToken(null); // token expired or revoked: forget it
      })
      .finally(() => setReady(true));
  }, []);

  // Save the token and remember the user
  const startSession = useCallback((r: AuthResult) => {
    writeToken(r.token);
    setUser(r.user);
  }, []);

  const register = useCallback(
    async (input: { name: string; email: string; phone?: string; password: string }) => {
      startSession(await registerRequest(input));
    },
    [startSession]
  );

  const login = useCallback(
    async (email: string, password: string) => {
      startSession(await loginRequest({ email, password }));
    },
    [startSession]
  );

  const logout = useCallback(async () => {
    const token = readToken();
    try {
      if (token) await logoutRequest(token); // revoke it on the server
    } catch {
      /* already invalid or offline: we still sign out here */
    }
    writeToken(null);
    setUser(null);
  }, []);

  const value = useMemo(
    () => ({ user, ready, register, login, logout }),
    [user, ready, register, login, logout]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

// The one-line way for any component to use it: const { user } = useAuth();
export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

===== src\lib\authApi.ts =====

// Talks to the Laravel auth endpoints. Every page and component goes through this file.

const BASE = process.env.NEXT_PUBLIC_API_URL ?? "http://127.0.0.1:8000/api/v1";

// A customer, exactly as UserResource sends it
export type User = {
  id: number;
  name: string;
  email: string;
  phone: string | null;
  createdAt: string;
};

// What register and login return
export type AuthResult = { user: User; token: string };

// A failed request. "fields" holds Laravel's per-field messages, e.g. { email: ["The email has already been taken."] }
export class ApiError extends Error {
  status: number; // HTTP status (0 = the server could not be reached)
  fields: Record<string, string[]>;

  constructor(message: string, status: number, fields: Record<string, string[]> = {}) {
    super(message);
    this.status = status;
    this.fields = fields;
  }
}

// One helper for every call: adds the JSON headers and the token, and turns failures into ApiError
  export async function call<T>(
  path: string,
  method: "GET" | "POST",
  token: string | null = null,
  body?: unknown,
  extraHeaders: Record<string, string> = {} // e.g. the Idempotency-Key
    ): Promise<T> {  
  let res: Response;
  try {
    res = await fetch(`${BASE}${path}`, {
      method,
      headers: {
        Accept: "application/json", // ask Laravel for JSON, even for errors
        ...(body ? { "Content-Type": "application/json" } : {}),
        ...(token ? { Authorization: `Bearer ${token}` } : {}), // proves who we are
        ...extraHeaders,
      },
      body: body ? JSON.stringify(body) : undefined,
      cache: "no-store", // never reuse an old answer for account data
    });
  } catch {
    throw new ApiError("Cannot reach the server. Check your connection and try again.", 0);
  }

  if (res.status === 204) return undefined as T; // logout answers with no body

  const data = await res.json().catch(() => ({})); // an HTML error page is not JSON
  if (!res.ok) {
    throw new ApiError(data.message ?? "Something went wrong. Please try again.", res.status, data.errors ?? {});
  }
  return data as T;
}

export const registerRequest = (input: { name: string; email: string; phone?: string; password: string }) =>
  call<AuthResult>("/register", "POST", null, input);

export const loginRequest = (input: { email: string; password: string }) =>
  call<AuthResult>("/login", "POST", null, input);

export const meRequest = (token: string) => call<{ user: User }>("/me", "GET", token);

export const logoutRequest = (token: string) => call<void>("/logout", "POST", token);

===== src\components\LoginForm.tsx =====

"use client"; // uses state, the router and Framer Motion

import { useEffect, useState } from "react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { motion } from "framer-motion";
import { LogIn } from "lucide-react";
import { useAuth } from "@/context/AuthContext";
import { ApiError } from "@/lib/authApi";

export default function LoginForm() {
  const router = useRouter();
  const { user, login } = useAuth();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState(""); // message shown under the button
  const [busy, setBusy] = useState(false); // true while we wait for the API

  // Signed in (just now, or already before)? Go to the account page.
  useEffect(() => {
    if (user) router.replace("/account");
  }, [user, router]);

  const submit = async (e: React.FormEvent) => {
    e.preventDefault(); // stop the page from reloading
    setError("");
    setBusy(true);
    try {
      await login(email.trim(), password);
      // success: "user" is now set and the effect above redirects
    } catch (err) {
      setError(err instanceof ApiError ? (err.fields.email?.[0] ?? err.message) : "Something went wrong.");
      setBusy(false);
    }
  };

  // Same look as the contact form
  const input =
    "w-full rounded-xl border border-ink/15 bg-white px-4 py-3.5 text-sm outline-none transition placeholder:text-muted focus:border-accent focus:ring-4 focus:ring-accent/10";
  const label = "mb-1.5 block text-xs font-semibold tracking-wide uppercase";

  return (
    <form onSubmit={submit} className="rounded-3xl border border-black/5 bg-white p-6 shadow-xl sm:p-10">
      <div className="space-y-5">
        <div>
          <label htmlFor="email" className={label}>Email</label>
          <input id="email" type="email" required autoComplete="email" className={input} placeholder="ada@email.com"
            value={email} onChange={(e) => setEmail(e.target.value)} />
        </div>
        <div>
          <label htmlFor="password" className={label}>Password</label>
          <input id="password" type="password" required autoComplete="current-password" className={input}
            value={password} onChange={(e) => setPassword(e.target.value)} />
        </div>

        <motion.button
          type="submit"
          disabled={busy}
          whileHover={{ scale: busy ? 1 : 1.02 }}
          whileTap={{ scale: busy ? 1 : 0.98 }}
          className="inline-flex w-full items-center justify-center gap-2 rounded-full bg-accent px-8 py-4 text-sm font-semibold tracking-[0.15em] text-white uppercase shadow-lg disabled:opacity-60"
        >
          {busy ? "Signing in..." : "Sign in"} <LogIn size={16} />
        </motion.button>

        {/* Error area. min-h reserves space so the form doesn't jump. */}
        <p role="alert" className="min-h-5 text-sm text-red-600">{error}</p>
      </div>

      <p className="mt-2 text-center text-sm text-muted">
        New here?{" "}
        <Link href="/register" className="font-semibold text-ink underline underline-offset-4 transition-colors hover:text-accent">
          Create an account
        </Link>
      </p>
    </form>
  );
}

===== src\components\AccountPanel.tsx =====

"use client"; // uses state from the login and the router

import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { LogOut} from "lucide-react";
import { useAuth } from "@/context/AuthContext";
import AccountOrders from "@/components/AccountOrders";


export default function AccountPanel() {
  const router = useRouter();
  const { user, ready, logout } = useAuth();

  // Once we know nobody is signed in, send them to the sign-in page
  useEffect(() => {
    if (ready && !user) router.replace("/login");
  }, [ready, user, router]);

  if (!ready || !user) return <p className="text-muted">Loading...</p>;

  const signOut = async () => {
    await logout();
    router.push("/");
  };

  // One row of the details card
  const row = (label: string, value: string) => (
    <div className="flex flex-wrap justify-between gap-2 border-b border-black/10 py-4 last:border-0">
      <dt className="text-sm text-muted">{label}</dt>
      <dd className="text-sm font-semibold">{value}</dd>
    </div>
  );

  return (
    <div className="grid max-w-4xl gap-8 md:grid-cols-2">
      {/* Details */}
      <section className="rounded-3xl border border-black/5 bg-white p-6 shadow-xl sm:p-8">
        <h2 className="text-xl font-semibold">Your details</h2>
        <dl className="mt-4">
          {row("Name", user.name)}
          {row("Email", user.email)}
          {row("Phone", user.phone ?? "Not added")}
        </dl>
        <button
          onClick={signOut}
          className="mt-6 inline-flex items-center gap-2 rounded-full border border-ink/15 px-6 py-3 text-sm font-semibold transition hover:border-ink"
        >
          <LogOut size={16} /> Sign out
        </button>
      </section>

           {/* Orders: loaded from the API */}
      <AccountOrders />
    </div>
  );
}
