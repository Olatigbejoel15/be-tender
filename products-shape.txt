class ProductWriteRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        $create = $this->isMethod('POST');
        $req = $create ? 'required' : 'sometimes'; // Create needs everything; edit only what is sent

        return [
            'version'          => $create ? ['prohibited'] : ['required', 'integer', 'min:1'], // Edits must say which version they saw
            'category_id'      => [$req, 'integer', 'exists:categories,id'],
            'name'             => [$req, 'string', 'max:120'],
            'price'            => [$req, 'integer', 'min:100', 'max:10000000'],                // Naira, whole numbers
            'badge'            => ['nullable', Rule::enum(ProductBadge::class)],
            'description'      => [$req, 'string', 'max:2000'],
            'features'         => ['nullable', 'array', 'max:10'],
            'features.*'       => ['string', 'max:120'],
            'fabric'           => [$req, 'string', 'max:255'],
            'care'             => [$req, 'string', 'max:500'],
            'size_label'       => ['nullable', Rule::enum(SizeLabel::class)],
            'sizes'            => [$req, 'array', 'min:1', 'max:12'],
            'sizes.*'          => ['string', 'max:20', 'distinct'],
            'sold_out_sizes'   => ['nullable', 'array'],
            'sold_out_sizes.*' => ['string', 'max:20'],
            'colors'           => [$req, 'array', 'min:1', 'max:20'],
            'colors.*.name'    => ['required', 'string', 'max:40', 'distinct:ignore_case'],
            'colors.*.hex'     => ['required', 'regex:/^#[0-9a-fA-F]{6}$/'],                    // e.g. #1a1a1f

            'is_published'     => ['sometimes', 'boolean'],
        ];

<?php

namespace App\Http\Resources;

use App\Models\ProductStock;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;
use Illuminate\Support\Str;

class ProductResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        $local = $this->images ?? [];                                   // Old local photo paths (fallback)
        $photos = $this->relationLoaded('photos') ? $this->photos : collect(); // Cloudinary rows
        $byColor = $photos->whereNotNull('color')->pluck('url', 'color');       // "black" => url
        $gallery = $photos->whereNull('color')->sortBy('position')->pluck('url')->values()->all();

        $colors = collect($this->colors ?? [])->map(fn ($c) => [
            ...$c,                                                      // name, hex, old image
            'image' => $byColor[Str::slug($c['name'])] ?? ($c['image'] ?? null), // Cloudinary first
        ])->all();

        $main = $gallery[0] ?? $byColor['black'] ?? $byColor->first()   // Main, else black, else any colour
            ?? ($local[0] ?? '/products/placeholder.svg');

        [$soldOutByColor, $soldOutSizes, $allSoldOut] = $this->availability();

        return [
            'id'             => $this->id,
            'slug'           => $this->slug,
            'name'           => $this->name,
            'category'       => $this->whenLoaded('category', fn () => $this->category->name),
            'price'          => $this->price,
            'image'          => $main,
            'gallery'        => array_slice($gallery, 1),               // Photos after the main one
            'badge'          => $this->badge?->value,                   // enum -> plain text, or null
            'description'    => $this->description,
            'features'       => $this->features ?? [],
            'fabric'         => $this->fabric,
            'care'           => $this->care,
            'sizeLabel'      => $this->size_label?->value,              // enum -> plain text
            'sizes'          => $this->sizes,
            'soldOut'        => $soldOutSizes,                          // Sizes unavailable in EVERY colour
            'soldOutByColor' => (object) $soldOutByColor,               // {"Black": ["XL"], ...} (counts are never exposed)
            'colors'         => $colors,
            'isSoldOut'      => $this->is_sold_out || $allSoldOut,      // Manual switch, or nothing left at all
            'isFeatured'     => $this->is_featured,
        ];
    }

    // Which options can't be bought right now: [by colour, sold-out sizes, everything sold out?]
    private function availability(): array
    {
        $manual = $this->sold_out_sizes ?? [];                          // Sizes an admin switched off by hand

        if (! $this->relationLoaded('stocks')) {
            return [[], $manual, false];                                // Stock not loaded: fall back to the manual flags
        }

        $qty = $this->stocks->mapWithKeys(                              // key => quantity
            fn ($s) => [ProductStock::key($s->product_id, $s->color, $s->size) => $s->quantity]
        );

        $colors = $this->colors ?? [];
        $sizes  = $this->sizes ?? [];
        $byColor = [];
        $outCount = 0;

        foreach ($colors as $c) {
            foreach ($sizes as $size) {
                $out = ($qty[ProductStock::key($this->id, $c['name'], $size)] ?? 0) <= 0 || in_array($size, $manual, true);

                if ($out) {
                    $byColor[$c['name']][] = $size;
                    $outCount++;
                }
            }
        }

        $soldOutSizes = array_values(array_filter($sizes, function ($size) use ($byColor, $colors) {
            return $colors !== [] && count(array_filter($byColor, fn ($list) => in_array($size, $list, true))) === count($colors);
        }));

        return [$byColor, $soldOutSizes, $colors !== [] && $outCount === count($colors) * count($sizes)];
    }
}

namespace App\Http\Controllers\Api\V1\Admin;

use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\ProductIndexRequest;
use App\Http\Requests\Admin\ProductWriteRequest;
use App\Http\Requests\Admin\StoreProductPhotoRequest;
use App\Http\Resources\AdminProductResource;
use App\Services\AdminProductService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Response;
use Illuminate\Http\Resources\Json\AnonymousResourceCollection;

class ProductController extends Controller
{
    public function __construct(private AdminProductService $products) {}

    // GET /api/v1/admin/products
    public function index(ProductIndexRequest $request): AnonymousResourceCollection
    {
        return AdminProductResource::collection($this->products->list($request->filters()));
    }

    // GET /api/v1/admin/products/{id}
    public function show(int $id): AdminProductResource
    {
        return new AdminProductResource($this->products->find($id));
    }

    // POST /api/v1/admin/products
    public function store(ProductWriteRequest $request): JsonResponse
    {
        $product = $this->products->create($request->user(), $request->payload());

        return (new AdminProductResource($product))->response()->setStatusCode(201); // 201 = created
    }

    // PATCH /api/v1/admin/products/{id}
    public function update(ProductWriteRequest $request, int $id): AdminProductResource
    {
        return new AdminProductResource(
            $this->products->update($request->user(), $id, $request->payload(), $request->version())
        );
    }

    // DELETE /api/v1/admin/products/{id}
    public function destroy(Request $request, int $id): Response
    {
        $this->products->delete($request->user(), $id);

        return response()->noContent(); // 204 = done, nothing to return
    }

    // POST /api/v1/admin/products/{id}/photos (multipart form)
    public function storePhoto(StoreProductPhotoRequest $request, int $id): AdminProductResource
    {
        return new AdminProductResource($this->products->addPhoto(
            $request->user(), $id, $request->file('photo'), $request->validated('color'), $request->validated('position')
        ));
    }

    // DELETE /api/v1/admin/products/{id}/photos/{photoId}
    public function destroyPhoto(Request $request, int $id, int $photoId): AdminProductResource
    {
        return new AdminProductResource($this->products->removePhoto($request->user(), $id, $photoId));
    }
}
